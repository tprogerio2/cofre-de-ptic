  <script>

    // SENHA FIXA
    const senhaCorreta = "12311111";

    const successSound = document.getElementById("successSound");
    const errorSound = document.getElementById("errorSound");

    const display = document.getElementById("display");
    const message = document.getElementById("message");
    const safe = document.getElementById("safe");
    const modal = document.getElementById("successModal");

    function addNumber(num){
      if(display.value.length < 8){
        display.value += num;
      }
    }

    function clearDisplay(){
      display.value = "";
      message.textContent = "";
    }

    function checkPassword(){

      if(display.value.length !== 8){

        message.style.color = "#ffcc00";
        message.textContent = "Digite 8 dígitos";

        return;
      }

      if(display.value === senhaCorreta){

        message.style.color = "#00ff90";
        message.textContent = "Cofre desbloqueado!";

        safe.classList.add("open");

        // ABRIR MODAL
        setTimeout(() => {
          modal.classList.add("show");
        }, 1000);
        successSound.play();

      }else{
         errorSound.play();
        message.style.color = "#ff4d4d";
        message.textContent = "Senha incorreta";

        display.value = "";

        safe.style.animation = "shake 0.3s";
       

        setTimeout(() => {
          safe.style.animation = "";
        }, 300);
      }
    }

    function closeModal(){
      modal.classList.remove("show");
    }

  </script>
